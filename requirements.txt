# ============================================
# RAG 문서 Q&A 엔진 - 필요 라이브러리 목록
# ============================================
# 설치 방법 (Ubuntu 서버에서 실행):
#   pip install -r requirements.txt
#
# 설치 후 버전 고정 방법:
#   pip freeze > requirements.txt
#   (잘 돌아가는 걸 확인한 뒤 실행하면 버전이 기록됨)
# ============================================

# --- PDF 파싱 ---
pdfplumber                    # PDF에서 텍스트/테이블 추출
PyPDF2                        # PDF 페이지별 텍스트 추출 (백업용)

# --- 텍스트 청킹 + RAG 파이프라인 ---
langchain                     # RAG 파이프라인 프레임워크
langchain-community           # LangChain 커뮤니티 통합 모듈
langchain-text-splitters      # 텍스트 청킹 (RecursiveCharacterTextSplitter 등)
langchain-openai              # OpenAI 임베딩/LLM 연동
langchain-anthropic           # Claude API 연동
langchain-google-genai        # Gemini API 연동
langchain-experimental        # Semantic Chunking 사용

# --- 임베딩 + 벡터 DB ---
chromadb                      # 벡터 데이터베이스 (임베딩 저장/검색)
openai                        # OpenAI API (임베딩 모델 호출)

# --- 로컬 임베딩/LLM ---
# Ollama를 사용하므로 torch, transformers, sentence-transformers는 불필요
# Ollama가 모델 로드/실행/메모리 관리를 모두 처리함
# 설치: curl -fsSL https://ollama.com/install.sh | sh

# --- UI ---
streamlit                     # 웹 인터페이스 (프론트+백엔드 통합)

# --- 유틸리티 ---
python-dotenv                 # .env 파일에서 API 키 로드
tiktoken                      # 토큰 수 계산 (청킹 크기 조절용)
requests                      # HTTP 요청 (Google API, Ollama API 호출용)
python-docx
python-pptx
openpyxl
